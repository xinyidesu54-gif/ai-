<script>
    //疼痛滑块
    const painSlider = document.getElementById('painScore');
    painSlider.addEventListener('input', ()=>{
        document.getElementById('painText').innerText = `当前：${painSlider.value}分`
    })

    //AI生成康复方案
    function generatePlan(){
        const part = document.getElementById('injurePart').value;
        const time = document.getElementById('injureTime').value;
        const pain = painSlider.value;
        const swell = document.getElementById('swell').value;
        let resultText = "";
        if(part === "shoulder"){
            if(time === "early"){
                resultText = `<p class="mb-3"><strong>诊断参考：肩部复合体损伤 急性期方案</strong></p>
                <p>✅ 优先控制炎症，禁止大幅度抬肩。</p>
                <p>✅ 推荐动作：钟摆运动、肩胛骨等长收缩。</p>
                <p>❌ 禁止负重、高举手臂，避免肩袖二次拉伤。</p>
                <p>💡 疼痛超过5分立刻停止训练，冰敷10分钟。</p>`
            }else if(time === "mid"){
                resultText = `<p class="mb-3"><strong>诊断参考：肩部复合体损伤 亚急性期方案</strong></p>
                <p>✅ 逐步增加肩关节被动活动范围。</p>
                <p>✅ 动作：钟摆、爬墙训练、外旋等长收缩。</p>
                <p>💡 训练以酸胀为限，不能产生尖锐刺痛。</p>`
            }else{
                resultText = `<p class="mb-3"><strong>诊断参考：肩部复合体损伤 恢复期方案</strong></p>
                <p>✅ 强化肩袖、斜方肌中下束，恢复肩部力量与活动度。</p>
                <p>✅ 动作：弹力带外旋、YTWL训练。</p>
                <p>💡 改善体态，预防圆肩代偿。</p>`
            }
        }else if(part === "wrist"){
            resultText = `<p><strong>腕骨骨折康复方案</strong></p><p>急性期以手指活动消肿为主；恢复期循序渐进做腕屈伸，避免过早负重。</p>`
        }else if(part === "knee"){
            resultText = `<p><strong>膝关节康复方案</strong></p><p>早期踝泵、股四头肌等长收缩；后期靠墙静蹲，训练时膝盖不超过脚尖。</p>`
        }else{
            resultText = `<p>已生成基础康复方案，请遵从康复师指导。训练中出现剧痛立刻停止。</p>`
        }
        document.getElementById('planText').innerHTML = resultText;
        document.getElementById('planResult').classList.remove('hidden');
    }

    //打卡记录
    let logs = [];
    function addLog(){
        const content = document.getElementById('trainLog').value;
        if(!content) return;
        const date = new Date().toLocaleString();
        logs.push(`【${date}】${content}`);
        renderLog();
        document.getElementById('trainLog').value = "";
    }
    function renderLog(){
        const box = document.getElementById('logList');
        box.innerHTML = logs.map(item=>`<div class="p-4 bg-slate-100 rounded-xl text-lg">${item}</div>`).join("");
    }

    //AI聊天
    function sendChat(){
        const msg = document.getElementById('chatInput').value;
        if(!msg) return;
        const chatBox = document.getElementById('chatBox');
        chatBox.innerHTML += `<div class="text-blue-700 my-2">用户：${msg}</div>`;
        document.getElementById('chatInput').value = "";
        setTimeout(()=>{
            let reply = "";
            if(msg.includes("肩") || msg.includes("肩部")){
                reply = "AI康复师：肩部训练切记不要耸肩代偿！钟摆运动身体要弯腰，手臂自然下垂摆动，不要主动发力抬胳膊。训练如果出现放射痛，马上停止。";
            }else if(msg.includes("疼痛") || msg.includes("痛")){
                reply = "AI康复师：训练后轻微酸胀属于正常；如果出现尖锐刺痛、夜间自发疼痛加重，建议暂停训练并联系康复师。急性期可冰敷，慢性酸痛可热敷。";
            }else if(msg.includes("骨折")){
                reply = "AI康复师：骨折康复优先保证骨愈合，不要过早负重。先做相邻关节活动，预防肌肉萎缩，严格遵从复查拍片结果。";
            }else{
                reply = "AI康复师：你可以描述你的损伤部位、受伤时间和训练遇到的问题，我帮你给出居家康复建议，注意本建议仅课程演示，不能替代线下康复师。";
            }
            chatBox.innerHTML += `<div class="text-emerald-700 my-2">AI康复师：${reply}</div>`;
            chatBox.scrollTop = chatBox.scrollHeight;
        },600)
    }
</script>
